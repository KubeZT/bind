#!/usr/bin/env bash main() ( # Set up access to a KubeZT cluster. Run with bash; do not use sudo. # Usage: source <(curl -fsSL https://bind.kubezt.com) CLUSTER set +x set -euo pipefail umask 077 if [ "$#" -ne 1 ]; then echo "Usage: source <(curl -fsSL https://bind.kubezt.com) CLUSTER" >&2 exit 1 fi CLUSTER_NAME=$1 [[ "$CLUSTER_NAME" =~ ^[a-z0-9][a-z0-9-]*$ ]] && [ "${#CLUSTER_NAME}" -le 48 ] || { echo "Invalid cluster name." >&2 exit 1 } PROFILE=$CLUSTER_NAME REGION=${AWS_REGION:-${AWS_DEFAULT_REGION:-us-gov-west-1}} KUBEZT_DIR=${KUBEZT_HOME:-$HOME/.kubezt} CLUSTER_DIR="$KUBEZT_DIR/clusters/$CLUSTER_NAME" BUCKET="kubezt-$CLUSTER_NAME-secrets" export AWS_SHARED_CREDENTIALS_FILE=${AWS_SHARED_CREDENTIALS_FILE:-$HOME/.aws/credentials} CREDENTIALS_FILE=$AWS_SHARED_CREDENTIALS_FILE AWS_DIR=$(dirname "$CREDENTIALS_FILE") mkdir -p "$CLUSTER_DIR" "$AWS_DIR" chmod 0700 "$KUBEZT_DIR" "$KUBEZT_DIR/clusters" "$CLUSTER_DIR" "$AWS_DIR" KUBEZT_DIR=$(cd "$KUBEZT_DIR" && pwd -P) CLUSTER_DIR="$KUBEZT_DIR/clusters/$CLUSTER_NAME" TOOLS_BIN="$KUBEZT_DIR/tools/bin" TEMP_DIR=$(mktemp -d "$CLUSTER_DIR/.download.XXXXXXXX") CREDENTIALS_LOCK='' cleanup() { rm -rf "$TEMP_DIR" if [ -n "$CREDENTIALS_LOCK" ]; then rmdir "$CREDENTIALS_LOCK" 2>/dev/null || true fi } trap cleanup EXIT trap 'exit 130' INT trap 'exit 143' TERM download() { curl --fail --silent --show-error --location \ --proto '=https' --proto-redir '=https' "$1" --output "$2" [ -s "$2" ] || { echo "Downloaded file is empty: $1" >&2; exit 1; } } verify_sha256() { local expected="$1" actual [[ "$expected" =~ ^[a-fA-F0-9]{64}$ ]] || { echo "Invalid SHA-256 checksum for $(basename "$2")." >&2 exit 1 } if command -v sha256sum >/dev/null 2>&1; then actual=$(sha256sum "$2" | awk '{print $1}') else actual=$(shasum -a 256 "$2" | awk '{print $1}') fi [ "$actual" = "$expected" ] || { echo "SHA-256 verification failed for $(basename "$2")." >&2 exit 1 } } check_destination() { if [ -L "$1" ] || { [ -e "$1" ] && [ ! -f "$1" ]; }; then echo "Destination is not a regular file: $1" >&2 exit 1 fi } check_credentials_file() { if [ -L "$CREDENTIALS_FILE" ] || { [ -e "$CREDENTIALS_FILE" ] && { [ ! -f "$CREDENTIALS_FILE" ] || [ ! -r "$CREDENTIALS_FILE" ]; }; }; then echo "AWS credentials must be a readable regular file: $CREDENTIALS_FILE" >&2 exit 1 fi } profile_exists() { [ -f "$CREDENTIALS_FILE" ] || return 1 awk -v profile="$PROFILE" ' /^[[:space:]]*\[/ { section = $0 sub(/^[[:space:]]*\[/, "", section) sub(/\].*$/, "", section) if (section == profile) found = 1 } END { exit !found } ' "$CREDENTIALS_FILE" } check_credentials_file if profile_exists; then echo "Using existing AWS profile [$PROFILE]." else # stdin contains the script when invoked through curl | bash. if ! { exec 3<>/dev/tty; } 2>/dev/null; then echo "AWS profile [$PROFILE] is missing. Run bind from an interactive terminal to enter credentials." >&2 exit 1 fi printf 'AWS credentials for [%s]\nAccess Key ID: ' "$PROFILE" >&3 IFS= read -r ACCESS_KEY_ID <&3 || exit 1 printf 'Access Key Secret: ' >&3 IFS= read -r -s ACCESS_KEY_SECRET <&3 || { printf '\n' >&3; exit 1; } printf '\n' >&3 exec 3>&- if ! [[ "$ACCESS_KEY_ID" =~ ^[A-Za-z0-9]+$ ]] || ! [[ "$ACCESS_KEY_SECRET" =~ ^[A-Za-z0-9/+=]+$ ]]; then echo "Credentials must be nonempty and contain only valid key characters." >&2 exit 1 fi # Serialize writes by bind, then recheck in case another run added the profile. if ! mkdir "$CREDENTIALS_FILE.kubezt-lock" 2>/dev/null; then echo "AWS credentials are locked by another bind run. Retry after it finishes: $CREDENTIALS_FILE.kubezt-lock" >&2 exit 1 fi CREDENTIALS_LOCK="$CREDENTIALS_FILE.kubezt-lock" check_credentials_file if profile_exists; then echo "Using existing AWS profile [$PROFILE]; entered credentials were not saved." else : >> "$CREDENTIALS_FILE" chmod 0600 "$CREDENTIALS_FILE" # Bash printf keeps credentials out of external process arguments. printf '\n[%s]\naws_access_key_id = %s\naws_secret_access_key = %s\n' \ "$PROFILE" "$ACCESS_KEY_ID" "$ACCESS_KEY_SECRET" >> "$CREDENTIALS_FILE" echo "Created AWS profile [$PROFILE] in $CREDENTIALS_FILE." fi unset ACCESS_KEY_ID ACCESS_KEY_SECRET rmdir "$CREDENTIALS_LOCK" CREDENTIALS_LOCK='' fi AWS_BIN=$(command -v aws || true) if [ -z "$AWS_BIN" ]; then AWS_BIN="$KUBEZT_DIR/tools/bin/aws" if [ ! -x "$AWS_BIN" ]; then echo "Installing AWS CLI..." curl --fail --silent --show-error --location \ --proto '=https' --proto-redir '=https' \ https://awscli.amazonaws.com/v2/install.sh \ --output "$TEMP_DIR/aws-install.sh" XDG_DATA_HOME="$KUBEZT_DIR/tools" XDG_BIN_HOME="$KUBEZT_DIR/tools/bin" \ bash "$TEMP_DIR/aws-install.sh" &2; exit 1; } fi fi ARTIFACTS=( "$CLUSTER_NAME-topology.json" "$CLUSTER_NAME" "$CLUSTER_NAME.pub" "$CLUSTER_NAME.config" ) for artifact in "${ARTIFACTS[@]}"; do echo "Downloading $artifact..." AWS_PAGER='' AWS_CLI_AUTO_PROMPT=off "$AWS_BIN" \ --profile "$PROFILE" --region "$REGION" \ s3 cp "s3://$BUCKET/$artifact" "$TEMP_DIR/$artifact" --only-show-errors [ -s "$TEMP_DIR/$artifact" ] || { echo "Empty artifact: $artifact" >&2; exit 1; } done echo "Downloading kubezt.sh..." download https://bind.kubezt.com/kubezt.sh "$TEMP_DIR/kubezt.sh" case "$(head -n 1 "$TEMP_DIR/kubezt.sh")" in '#!/bin/bash'|'#!/usr/bin/env bash') ;; *) echo "The downloaded kubezt.sh is not a Bash script." >&2; exit 1 ;; esac bash -n "$TEMP_DIR/kubezt.sh" chmod 0700 "$TEMP_DIR/kubezt.sh" case "$(uname -s)" in Darwin) PLATFORM=darwin; JQ_PLATFORM=macos ;; Linux) PLATFORM=linux; JQ_PLATFORM=linux ;; *) echo "Supported platforms: macOS and Linux." >&2; exit 1 ;; esac case "$(uname -m)" in arm64|aarch64) ARCH=arm64 ;; x86_64|amd64) ARCH=amd64 ;; *) echo "Supported architectures: amd64 and arm64." >&2; exit 1 ;; esac mkdir -p "$TOOLS_BIN" chmod 0700 "$KUBEZT_DIR/tools" "$TOOLS_BIN" JQ_BIN=$(command -v jq || true) if [ -z "$JQ_BIN" ]; then JQ_BIN="$TOOLS_BIN/jq" if [ ! -x "$JQ_BIN" ]; then echo "Installing jq to read the cluster topology..." JQ_ASSET="jq-$JQ_PLATFORM-$ARCH" JQ_RELEASE=https://github.com/jqlang/jq/releases/download/jq-1.8.2 download "$JQ_RELEASE/$JQ_ASSET" "$TEMP_DIR/jq" download "$JQ_RELEASE/sha256sum.txt" "$TEMP_DIR/jq.sha256" JQ_SHA256=$(awk -v asset="$JQ_ASSET" '$2 == asset || $2 == ("*" asset) {print $1}' "$TEMP_DIR/jq.sha256") verify_sha256 "$JQ_SHA256" "$TEMP_DIR/jq" chmod 0700 "$TEMP_DIR/jq" check_destination "$JQ_BIN" mv -f "$TEMP_DIR/jq" "$JQ_BIN" fi fi KUBERNETES_VERSION=$("$JQ_BIN" -er --arg name "$CLUSTER_NAME" ' if type != "array" then error("Topology must be an array of clusters") else . end | [.[] | select(.name == $name)] | if length != 1 then error("Topology must contain exactly one matching cluster") else .[0] end | .version.rke2.version | if type != "string" then error("Missing cluster Kubernetes version") else . end ' "$TEMP_DIR/$CLUSTER_NAME-topology.json") KUBERNETES_VERSION=${KUBERNETES_VERSION#v} KUBERNETES_VERSION=${KUBERNETES_VERSION%%+*} case "$KUBERNETES_VERSION" in [0-9]*.[0-9]*.[0-9]*) ;; *) echo "Invalid Kubernetes version in the topology." >&2 exit 1 ;; esac SERVER_MAJOR=${KUBERNETES_VERSION%%.*} SERVER_REST=${KUBERNETES_VERSION#*.} SERVER_MINOR=${SERVER_REST%%.*} case "$SERVER_MAJOR:$SERVER_MINOR" in *[!0-9:]*|:|*:) echo "Invalid Kubernetes version in the topology." >&2 exit 1 ;; esac compatible_kubectl() { local version client_major client_rest client_minor minor_difference [ -x "$1" ] || return 1 version=$("$1" version --client --output=json 2>/dev/null | "$JQ_BIN" -er '.clientVersion.gitVersion') || return 1 version=${version#v} case "$version" in [0-9]*.[0-9]*.*) ;; *) return 1 ;; esac client_major=${version%%.*} client_rest=${version#*.} client_minor=${client_rest%%.*} case "$client_major:$client_minor" in *[!0-9:]*|:|*:) return 1 ;; esac [ "$client_major" = "$SERVER_MAJOR" ] || return 1 minor_difference=$((client_minor - SERVER_MINOR)) [ "$minor_difference" -ge -1 ] && [ "$minor_difference" -le 1 ] } KUBECTL_BIN="$TOOLS_BIN/kubectl" if ! compatible_kubectl "$KUBECTL_BIN"; then EXISTING_KUBECTL=$(command -v kubectl || true) if [ -n "$EXISTING_KUBECTL" ] && compatible_kubectl "$EXISTING_KUBECTL"; then echo "Using installed kubectl." # env.sh puts this directory on PATH for the administrator tool. ln -s "$EXISTING_KUBECTL" "$TEMP_DIR/kubectl" else echo "Installing kubectl v$KUBERNETES_VERSION..." KUBECTL_URL="https://dl.k8s.io/release/v$KUBERNETES_VERSION/bin/$PLATFORM/$ARCH/kubectl" download "$KUBECTL_URL" "$TEMP_DIR/kubectl" download "$KUBECTL_URL.sha256" "$TEMP_DIR/kubectl.sha256" verify_sha256 "$(awk '{print $1}' "$TEMP_DIR/kubectl.sha256")" "$TEMP_DIR/kubectl" chmod 0700 "$TEMP_DIR/kubectl" fi # This path may already be a symlink created by an earlier bind run. if [ -d "$KUBECTL_BIN" ]; then echo "kubectl destination is a directory: $KUBECTL_BIN" >&2 exit 1 fi mv -f "$TEMP_DIR/kubectl" "$KUBECTL_BIN" fi "$KUBECTL_BIN" --kubeconfig "$TEMP_DIR/$CLUSTER_NAME.config" \ config view --minify -o 'jsonpath={.current-context}' >/dev/null chmod 0600 "$TEMP_DIR/$CLUSTER_NAME-topology.json" \ "$TEMP_DIR/$CLUSTER_NAME" "$TEMP_DIR/$CLUSTER_NAME.config" chmod 0644 "$TEMP_DIR/$CLUSTER_NAME.pub" # Save the files only after all four downloads succeed. for artifact in "${ARTIFACTS[@]}"; do check_destination "$CLUSTER_DIR/$artifact" done check_destination "$KUBEZT_DIR/kubezt.sh" for artifact in "${ARTIFACTS[@]}"; do mv -f "$TEMP_DIR/$artifact" "$CLUSTER_DIR/$artifact" done mv -f "$TEMP_DIR/kubezt.sh" "$KUBEZT_DIR/kubezt.sh" echo "Artifacts saved to $CLUSTER_DIR" echo "KubeZT administrator tool installed at $KUBEZT_DIR/kubezt.sh" ) main "$@" || return $? export KUBEZT_HOME=${KUBEZT_HOME:-$HOME/.kubezt} KUBEZT_HOME=$(cd "$KUBEZT_HOME" && pwd -P) export CLUSTER_NAME=$1 export KUBECONFIG="$KUBEZT_HOME/clusters/$CLUSTER_NAME/$CLUSTER_NAME.config" export KUBEZT_ADMIN_SCRIPT="$KUBEZT_HOME/kubezt.sh" case ":${PATH-}:" in *":$KUBEZT_HOME/tools/bin:"*) ;; *) export PATH="$KUBEZT_HOME/tools/bin:${PATH-}" ;; esac unalias kubezt 2>/dev/null || true kubezt() { "$KUBEZT_ADMIN_SCRIPT" "$@" } if [ -n "${ZSH_VERSION-}" ]; then export PS1="KubeZT ($CLUSTER_NAME) [%1~]$ " else export PS1="KubeZT ($CLUSTER_NAME) [\W]$ " fi echo "KubeZT ($CLUSTER_NAME) is active." echo "Run: kubezt status"
